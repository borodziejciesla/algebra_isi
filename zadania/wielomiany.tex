\begin{enumerate}
    \item Wykonać dodawanie $(P+Q)(z) = P(z) + Q(z)$, odejmowanie $(P-Q)(z) = P(z) - Q(z)$ i~mnożenie $(P \cdot Q)(z) = P(z) \cdot Q(z)$ wielomianów:
    \begin{enumerate}
        \item $P(z) = 1 - z^{2}, Q(z) = -1 - 5z + z^{2}$,
        \item $P(z) = 2z^{5} - 3z^{2} + z, Q(z) = 2z^{2} + 2z + 3$,
        \item $P(z) = z^{2} + i, Q(z) = (1-i)z^{3} + iz + 3 - 2i$,
    \end{enumerate}

    \item Znaleźć pierwiastki wielomianów:
    \begin{enumerate}
        \item $P(z) = z^{2}+2iz+3$,
        \item $P(z) = z^{2}-(2+i)z-1+7i$,
        \item $P(z) = z^{4} - 30z^{2} + 289$,
    \end{enumerate} 

    \item Znaleźć pozostałe pierwiastki wielomianu znajac niektóre spośród ich pierwiastków:
    \begin{enumerate}
        \item $P(z) = z^{4}+2z^{3}+5z^{2}+6z+6, z_{1}=-1+i$,
        \item $P(z) = z^{5}-5z^{4}+18z^{3}-18z^{2}+17z-13, z_{1}=2-3i, z_{2}=i$,
    \end{enumerate} 
\end{enumerate}
